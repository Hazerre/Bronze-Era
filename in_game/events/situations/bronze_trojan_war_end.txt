namespace = trojan_war_end

trojan_war_end.1 = {
    type = country_event
    title = trojan_war_end_01.t
    desc = trojan_war_end_01.d
    is_triggered_only = yes
    hidden = no
# For Greece victory
    immediate = {
        root = { 
                add_prestige = 10
            OR = {
                if = { limit = { global_var:anatolia_raid_success_count = 1 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_one  
                    days = -1
                    }
                }
                if = { limit = { global_var:anatolia_raid_success_count = 2 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_two  
                    days = -1
                    }
                }
                if = { limit = { global_var:anatolia_raid_success_count = 3 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_three  
                    days = -1
                    }
                }
                if = { limit = { global_var:anatolia_raid_success_count = 4 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_four  
                    days = -1
                    }
                }
                if = { limit = { global_var:anatolia_raid_success_count >= 5 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_five  
                    days = -1
                    }
                }
            }
        }
    }
    option = {
        name = "trojan_war_end.a"
    }
}
