namespace = trojan_war_phase_two_events

trojan_war_phase_two_events.1 = {
    type = country_event
    title = trojan_war_phase_2_start.t
    desc = trojan_war_phase_2_start.d
    is_triggered_only = yes
    hidden = no

immediate = {
    declare_war_with_cb = {
        target = c:WILUS
        type = casus_belli:trojan_war_casus_belli
        set_variable = {
            name = is_trojan_war
            value = yes
        }
    }
    c:WILUS = { save_scope_as = target_country }
    c:0001G = { save_scope_as = target }
    random_current_war = {
        limit = {
            any_defender = { this = c:WILUS }
        }
        save_scope_as = trojan_war 
    }
    c:AHHI = { trigger_event_non_silently = { id = trojan_war_phase_two_events.2 days = 5 } }
    c:LYCIA = { trigger_event_non_silently = { id = trojan_war_phase_two_events.2 days = 5 } }
    c:SEHA = { trigger_event_non_silently = { id = trojan_war_phase_two_events.2 days = 5 } }
    c:LUKKA = { trigger_event_non_silently = { id = trojan_war_phase_two_events.2 days = 5 } }
    c:MASA = { trigger_event_non_silently = { id = trojan_war_phase_two_events.2 days = 5 } }
} 
option = {
    name = "trojan_war_phase_2_start.a"
} 
}
